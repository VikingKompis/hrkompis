        const removeId = $(this).data('recruitertype-id');

        let rtids = url.searchParams.get('rtids')
            ?.split(',')
            .map(rtidsStr => parseInt(rtidsStr)) || []; // Convert each to an integer

        rtids = rtids.filter(id => id !== removeId);

        // Remove duplicates
        rtids = [...new Set(rtids)];

        if (rtids.length > 0){
            url.searchParams.set('rtids', rtids);
        }
        else{
            url.searchParams.delete('rtids');
        }

        // Remove 'pagenumber' query parameter if it exists
        url.searchParams.delete('pagenumber');

        // Update the URL without reloading immediately
        window.history.pushState({}, '', url);
        // Reload the page to apply the filter
        location.reload();
    });

        }

        const slug = this.getAttribute('data-slug');
        const url = new URL(window.location.href);

        // Handle query parameters
        const searchParams = new URLSearchParams(url.search);

        // Manage path segments for slugs
        let pathSegments = url.pathname.split('/').filter(segment => segment).map(segment => decodeURIComponent(segment));

        if (enableAdvancedLocationSearch && $(this).hasClass('active')) {
            pathSegments = pathSegments.filter(s => s !== slug);
        } else {
            if (!pathSegments.includes(slug)) {
                pathSegments.push(slug);
            }
        }

        // Update the URL path and preserve query params
        url.pathname = '/' + pathSegments.join('/');
        url.search = searchParams.toString();

        // Remove 'pagenumber' query parameter if it exists
        url.searchParams.delete('pagenumber');
        
        // Update the URL without reloading immediately
        window.history.pushState({}, '', url);
        // Reload the page to apply the filter
        location.reload();
    });

    function applyNow(productId, redirectUrl, btnElement) {
        var formData = new FormData();

        var $button = $(btnElement);
        formData.append('Id', productId);
        
        if ($button.data('processing')) {
            return;
        }
        $button.data('processing', true);
        $button.prop('disabled', true).text('Processing...');

        $.ajax({
            url: '/ApplyJob/ApplyNow',
            type: 'POST',
            data: formData,
            processData: false,
            contentType: false,
            success: function (response) {
                if (response.success) {
                    setTimeout(function () {
                        location.reload();
                    }, 100);
                } else {
                    if (!response.success && response.unauthorize) {
                        window.location.href = response.redirect;
                    } else if (!response.success) {
                        alert(response.message || "Something went wrong!");
                        resetButton();
                    }
                }
            },
            error: function (xhr, status, error) {
                console.error("AJAX Error:", xhr.responseText);
                alert("An error occurred while submitting the application.");
                resetButton();
            }
        });
    }

    function resetButton() {
        $button.data('processing', false);
        $button.prop('disabled', false).text('Apply Now'); // Restore button text
    }

    $(function() {
        $('.save-button').on('click', function(event) {

                var wasOpen = $popup.hasClass('show');
                $('.auth-popup.show').removeClass('show');

                if (!wasOpen) {
                    $popup.addClass('show');
                }
            } 
        });

        $(document).on('click', function(event) {
            if (!$(event.target).closest('.save-button').length &&

        });
