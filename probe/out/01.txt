    <script>
        (() => {
            // One dropdown open at a time; a click elsewhere closes them; the button
            // shows what is ticked.
            const all = () => document.querySelectorAll('[data-aviation-dropdown]');
            document.addEventListener('toggle', (event) => {
                const opened = event.target;
                if (!(opened instanceof HTMLDetailsElement) || !opened.matches('[data-aviation-dropdown]') || !opened.open) {
                    return;
                }
                all().forEach((other) => { if (other !== opened) { other.open = false; } });
            }, true);
            document.addEventListener('click', (event) => {
                all().forEach((dropdown) => {
                    if (dropdown.open && !dropdown.contains(event.target)) {
                        dropdown.open = false;
                    }
                });
            });
            document.addEventListener('change', (event) => {
                const box = event.target;
                const dropdown = box instanceof HTMLInputElement ? box.closest('[data-aviation-dropdown]') : null;
                if (!dropdown) {
                    return;
                }
                const ticked = [...dropdown.querySelectorAll('input[type="checkbox"]:checked')].map((input) => input.dataset.label);
                const value = dropdown.querySelector('[data-aviation-dropdown-value]');
                value.textContent = ticked.length === 0 ? 'Any' : ticked[0] + (ticked.length > 1 ? ' +' + (ticked.length - 1) : '');
                value.classList.toggle('is-set', ticked.length > 0);
            });
        })();
